import { appendFileSync, existsSync, writeFileSync } from 'node:fs';
import { expect, it } from 'vitest';
import { recovered } from './recovered.ts';

let poisoned = false;
it('recovers only in a fresh process', () => {
  expect(poisoned).toBe(false);
  poisoned = true;
  appendFileSync('attempts.txt', 'attempt\n');
  if (!existsSync('attempted.txt')) {
    writeFileSync('attempted.txt', 'failed');
    throw new Error('First attempt leaves module state behind');
  }
  expect(recovered()).toBe(2);
  expect(process.env.PERMANENT_FAILURE).not.toBe('true');
});
it('passes in the failed file', () => {
  appendFileSync('passed.txt', 'passed\n');
  expect(1).toBe(1);
});
