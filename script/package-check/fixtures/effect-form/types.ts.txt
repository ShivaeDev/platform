import { Effect, Layer, Schema } from "effect";
import * as Atom from "effect/reactivity/Atom";
import { form } from "@shivaedev/effect-form/form.ts";
import type { FieldFailure, Form } from "@shivaedev/effect-form/shape.ts";
import { useField, useSubmit } from "@shivaedev/effect-form/client/react.ts";
const schema = Schema.Struct({ count: Schema.NumberFromString });
const currentForm = form(schema, {
  initialValues: { count: "1" },
  runtime: Atom.runtime(Layer.empty),
  onSubmit: (values) => {
    const count: number = values.count;
    return Effect.succeed(count);
  },
});
const encoded: string = currentForm.field("count").value;
void encoded;
const message: Atom.Atom<string | undefined> = currentForm.errorAt(["count"]);
void message;
void (currentForm satisfies Form<typeof schema.fields, number, never, never>);
// @ts-expect-error Field names come from the schema.
currentForm.field("missing");
// @ts-expect-error Editing accepts encoded values.
currentForm.change("count", 1);
function Consumer() {
  const field = useField(currentForm, "count");
  const value: string = field.value;
  const submit = useSubmit(currentForm);
  const run: () => void = submit.run;
  void value;
  void run;
  // @ts-expect-error React field values retain the encoded type.
  field.onChange(12);
  return null;
}
void Consumer;

function failureField(failure: FieldFailure): string {
  return failure.path;
}
void failureField;

form(Schema.Struct({ answers: Schema.Array(Schema.Struct({ count: Schema.NumberFromString, id: Schema.String })) }), {
  arrayKeys: {
    answers: (answer) => {
      const count: string = answer.count;
      void count;
      // @ts-expect-error Key selectors receive encoded array elements.
      const decoded: number = answer.count;
      void decoded;
      return answer.id;
    },
  },
  initialValues: { answers: [] },
  onSubmit: (_values, submitter) => {
    // @ts-expect-error Rejection paths must begin with a schema field.
    submitter.fail(["missing", 0], "Rejected");
    return submitter.fail(["answers", 0, "count"], "Rejected");
  },
  runtime: Atom.runtime(Layer.empty),
});

form(schema, {
  arrayKeys: {
    // @ts-expect-error A scalar field cannot have an array key selector.
    count: () => "key",
  },
  initialValues: { count: "1" },
  onSubmit: Effect.succeed,
  runtime: Atom.runtime(Layer.empty),
});
