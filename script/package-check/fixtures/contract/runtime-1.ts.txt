import { Effect, Layer, Option, Schema, Stream } from 'effect';
import * as AtomRegistry from 'effect/reactivity/AtomRegistry';
import * as AtomRpc from 'effect/reactivity/AtomRpc';
import * as Reactivity from 'effect/reactivity/Reactivity';
import { RpcTest } from 'effect/rpc';
import { collection, invalidationKeys } from '@shivaedev/changes/keys.ts';
import { LiveConnection } from '@shivaedev/contract/automaticLive.ts';
import { bind } from '@shivaedev/contract/bind.ts';
import { changesHandlerLayer } from '@shivaedev/contract/changesHandlerLayer.ts';
import { contract } from '@shivaedev/contract/contract.ts';
import { makeLiveServer } from '@shivaedev/contract/makeLiveServer.ts';
import { query } from '@shivaedev/contract/operation.ts';
import { querySubscription } from '@shivaedev/contract/querySubscription.ts';

const workspace = collection('workspace', Schema.String);
const Api = contract('workspace', { queries: [query('read', {
 readPolicy: () => Effect.succeed(true), reads: () => [workspace.list], success: Schema.Struct({ title: Schema.String })
})] });
let resumeListener: (() => void) | undefined;
await Effect.runPromise(Effect.scoped(Effect.gen(function* () {
 const reactivity = yield* Reactivity.make;
 const server = yield* makeLiveServer.pipe(Effect.provideService(Reactivity.Reactivity, reactivity));
 let title = 'First';
 let runs = 0;
 const read = Effect.sync(() => { runs += 1; return { title }; });
 const handlers = Layer.merge(
  Api.application.toLayer({ 'workspace.read': () => read }),
  changesHandlerLayer(Api, rpc => ({ queries }) => Stream.unwrap(Effect.gen(function* () {
   const subscription = yield* querySubscription(rpc, queries, Option.none());
   return server.subscribe(subscription, () => read);
  })))
 );
 class Client extends AtomRpc.Service<Client>()('packed/Push', {
  group: Api, makeEffect: RpcTest.makeClient(Api, { flatten: true }), protocol: handlers
 }) {}
 const api = bind(Api, Client, { resume: listener => {
  resumeListener = listener;
  return () => { resumeListener = undefined; };
 } });
 const registry = yield* Effect.acquireRelease(Effect.sync(() => AtomRegistry.make()), current => Effect.sync(() => current.dispose()));
 const atom = api.read.query();
 registry.mount(atom);
 function shown(expected: string) {
  return AtomRegistry.toStreamResult(registry, atom).pipe(Stream.filter(value => value.title === expected), Stream.runHead);
 }
 yield* shown('First');
 if (runs !== 1) throw new Error('Packed subscription did not supply its first result in one read');
 title = 'Pushed';
 yield* reactivity.invalidate(invalidationKeys([workspace.list]));
 yield* shown('Pushed');
 registry.set(api[LiveConnection].paused, true);
 title = 'While paused';
 yield* reactivity.invalidate(invalidationKeys([workspace.list]));
 yield* AtomRegistry.toStream(registry, api[LiveConnection].status).pipe(Stream.filter(status => status.pending === 1), Stream.runHead);
 if ((yield* AtomRegistry.getResult(registry, atom)).title !== 'Pushed') throw new Error('Packed paused query replaced its visible result');
 registry.set(api[LiveConnection].paused, false);
 yield* shown('While paused');
 title = 'After resume';
 if (resumeListener === undefined) throw new Error('Packed resume source was not initialized');
 resumeListener();
 yield* shown('After resume');
})));
if (resumeListener !== undefined) throw new Error('Packed resume source was not released');
