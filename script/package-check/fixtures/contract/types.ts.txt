import { Effect, Schema, Stream } from "effect"
import { collection } from "@shivaedev/contract/keys.ts"
import { command, query, type QueryShape } from "@shivaedev/contract/operation.ts"
import { contract } from "@shivaedev/contract/contract.ts"
import { fieldRejection } from "@shivaedev/contract/rejection.ts"

class Missing extends Schema.TaggedError<Missing>()("Missing", {}) {}
const Draft = Schema.Struct({ title: Schema.String })
const notes = collection("notes", Schema.Number)
const Get = query("get", { payload: { id: Schema.Number }, success: Schema.String, rejections: { Missing }, reads: ({ id }) => [notes.item(id)] })
const Rename = command("rename", {
  payload: { id: Schema.Number, title: Schema.String },
  rejections: { Missing, Invalid: fieldRejection(Draft) },
  invalidates: ({ id }) => [notes.item(id)],
})
const Notes = contract("notes", { queries: [Get], commands: [Rename] })
export const handlers = Notes.of({
  "notes.$changes": () => Stream.never,
  "notes.get": ({ id }) => (id === 1 ? Effect.succeed("one") : Get.reject.Missing()),
  "notes.rename": ({ title }) => (title === "" ? Rename.reject.Invalid({ field: "title", message: "Enter a title" }) : Effect.void),
})
// @ts-expect-error Key ids keep the collection's id type through the packed declarations.
notes.item("1")
// @ts-expect-error Field rejections accept only the struct's keys.
Rename.reject.Invalid({ field: "body", message: "" })
const shapes: readonly QueryShape[] = [Get]
// @ts-expect-error QueryShape.reads keeps its type from @shivaedev/types through the packed declarations.
export const readsName: string = shapes[0]?.reads({ id: 1 })

import { stream } from "@shivaedev/contract/operation.ts"
import { bind } from "@shivaedev/contract/bind.ts"
import * as AtomRpc from "effect/reactivity/AtomRpc"
import { RpcTest } from "effect/rpc"
import type { RpcClientError } from "effect/rpc/RpcClientError"
const Subscribe = stream("subscribe", { payload: { id: Schema.Number }, success: Schema.String, rejections: { Missing }, reads: ({ id }) => [notes.item(id)] })
const Feed = contract("feed", { streams: [Subscribe] })
const feedHandlers = Feed.toLayer({ "feed.$changes": () => Stream.never, "feed.subscribe": ({ id }) => Stream.succeed(String(id)) })
class Client extends AtomRpc.Service<Client>()("packed/Stream", { group: Feed, makeEffect: RpcTest.makeClient(Feed, { flatten: true }), protocol: feedHandlers }) {}
const api = bind(Feed, Client)
export const subscription: Stream.Stream<string, Missing | RpcClientError, Client> = api.subscribe.run({ id: 1 })
// @ts-expect-error Stream payloads remain typed through the packed declarations.
api.subscribe.run({ id: "1" })
// @ts-expect-error Bound streams expose run, without a query atom.
api.subscribe.query()
// @ts-expect-error Literal stream operation names share the contract namespace.
contract("feed", { queries: [query("subscribe", { reads: () => [] })], streams: [Subscribe] })
