import { Deferred, Effect, Option } from "effect";
import * as Atom from "effect/reactivity/Atom";
import * as AtomRegistry from "effect/reactivity/AtomRegistry";
import { act, createElement } from "react";
import { expect, it } from "vitest";
import { useQuery } from "@shivaedev/react/client/resultState.ts";
import { expectNoReactWarnings, render, type Rendered } from "@shivaedev/test-story/client/react.ts";
import { screenDriver } from "@shivaedev/test-story/server/screenDriver.ts";

Object.assign(globalThis, { IS_REACT_ACT_ENVIRONMENT: true });
expectNoReactWarnings();

it("renders with the packed testing entry and releases the registry", async () => {
  let finalized = false;
  const resource = Atom.keepAlive(Atom.make((get) => {
    get.addFinalizer(() => { finalized = true; });
    return "ready";
  }));
  const answer = Effect.runSync(Deferred.make<string>());
  const query = Atom.make(Deferred.await(answer));
  function Component() {
    return createElement("output", null, Option.getOrNull(useQuery(query).data));
  }
  const view: Rendered = await render(createElement(Component));
  try {
    expect(view.registry.get(resource)).toBe("ready");
    const screen = screenDriver({
      act,
      pending: AtomRegistry.toStream(view.registry, Atom.map(query, (result) => result.waiting)),
      root: view.container,
      tell: () => undefined,
    });
    await act(async () => {
      await Effect.runPromise(Deferred.succeed(answer, "settled"));
    });
    await Effect.runPromise(screen.settle());
    expect(screen.text("output")).toBe("settled");
    await view.render(createElement("output", null, "rerendered"));
    expect(view.container.textContent).toBe("rerendered");
  } finally {
    await view.unmount();
  }
  expect(finalized).toBe(true);
  expect(view.container.isConnected).toBe(false);
});
