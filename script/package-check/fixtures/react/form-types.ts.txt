import { Data, Effect, Layer, type Option, Schema } from "effect";
import * as Atom from "effect/reactivity/Atom";
import { useCreate } from "@shivaedev/react/client/create.ts";
import { useEditor } from "@shivaedev/react/client/editor.ts";

const query = Atom.make(Effect.succeed({ id: 1, title: "Quarterly report" }));
const fields = Schema.Struct({ title: Schema.String });
const runtime = Atom.runtime(Layer.empty);
const saveTitle = (values: { readonly title: string }) =>
	values.title.length > 0 ? Effect.succeed({ id: 1, title: values.title }) : Effect.fail("Rejected" as const);

export const useEditorExample = () => {
	const editor = useEditor(
		{ run: saveTitle, runtime },
		// An untagged failure is the framework's to report, so it needs no handler.
		{ query, fields },
	);
	const failure: Option.Option<"Rejected"> = editor.failure;
	const row: Option.Option<{ id: number; title: string }> = editor.query.data;
	const create = useCreate(
		{ run: saveNamed, runtime },
		{
			fields,
			initialValues: { title: "" },
			// @ts-expect-error A handler returns a field rejection only for a declared field.
			failures: { NameRejected: () => ({ field: "missing", message: "Unknown" }) },
		},
	);
	const created: Option.Option<never> = create.created;
	return { failure, row, created, save: editor.save };
};

class NameRejected extends Data.TaggedError("NameRejected")<{ readonly field: "title"; readonly message: string }> {}
class StaleRejected extends Data.TaggedError("StaleRejected")<{ readonly field: "subtitle"; readonly message: string }> {}
const saveNamed = (values: { readonly title: string }) => Effect.fail(new NameRejected({ field: "title", message: values.title }));
const saveStale = (values: { readonly title: string }) => Effect.fail(new StaleRejected({ field: "subtitle", message: values.title }));

const saveStaleRow = (values: { readonly title: string }) => Effect.as(saveStale(values), { id: 1, title: values.title });

export const useRejectionExample = () => {
	const named = useCreate({ run: saveNamed, runtime }, { fields, initialValues: { title: "" } });
	// @ts-expect-error A tagged rejection naming a field the form lacks is the app's own, so it needs a failures handler.
	const stale = useCreate({ run: saveStale, runtime }, { fields, initialValues: { title: "" } });
	const mapped = useEditor(
		{ run: saveStaleRow, runtime },
		{ query, fields, failures: { StaleRejected: (error) => ({ field: "title", message: error.message }) } },
	);
	// @ts-expect-error The editor applies the same constraint to its save command.
	const unmapped = useEditor({ run: saveStaleRow, runtime }, { query, fields });
	return { named, stale, mapped, unmapped };
};

const perField = { message: Schema.String, field: Schema.optionalKey(Schema.String) };
class Conflict extends Schema.TaggedError<Conflict>()("Conflict", perField) {}
class TitleConflict extends Schema.TaggedError<TitleConflict>()("TitleConflict", {
	message: Schema.String,
	field: Schema.optionalKey(Schema.Literal("title")),
}) {}
class Described extends Schema.TaggedError<Described>()("Described", { message: Schema.String }) {}
const saveConflict = (values: { readonly title: string }) => Effect.fail(new Conflict({ message: values.title }));

const saveTitled = () => Effect.fail(new TitleConflict({ message: "Taken" }));
const saveDescribed = () => Effect.fail(new Described({ message: "Gone" }));

export const useOptionalFieldExample = () => {
	// A plain string field may name anything, so the default mapping handles it and reports other names as the save failure.
	const unmapped = useCreate({ run: saveConflict, runtime }, { fields, initialValues: { title: "" } });
	const mapped = useCreate(
		{ run: saveConflict, runtime },
		{
			fields,
			initialValues: { title: "" },
			failures: { Conflict: (error) => (error.field === "title" ? { field: "title", message: error.message } : { message: error.message }) },
		},
	);
	// The app's own rejections say what they show; "message" shows the error's message as a notice.
	const titled = useCreate({ run: saveTitled, runtime }, { fields, initialValues: { title: "" }, failures: { TitleConflict: "message" } });
	const described = useCreate({ run: saveDescribed, runtime }, { fields, initialValues: { title: "" }, failures: { Described: "message" } });
	// @ts-expect-error Leaving out the handler for an app-defined rejection does not compile.
	const unhandled = useCreate({ run: saveDescribed, runtime }, { fields, initialValues: { title: "" } });
	const quiet = useCreate({ run: saveDescribed, runtime }, { fields, initialValues: { title: "" }, silent: true });
	return { unmapped, mapped, titled, described, unhandled, quiet };
};
