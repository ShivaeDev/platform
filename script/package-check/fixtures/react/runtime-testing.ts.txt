import "happy-dom";
import { createVitest } from "vitest/node";

const runner = await createVitest("test", {
  config: false,
  environment: "happy-dom",
  include: ["render.dom.spec.ts"],
  root: process.cwd(),
  watch: false,
});
try {
  const { testModules, unhandledErrors } = await runner.start();
  const tests = testModules.flatMap((module) => [...module.children.allTests()]);
  if (unhandledErrors.length !== 0 || tests.length !== 1 || tests.some((test) => test.result().state !== "passed")) {
    throw new Error("The packed React render test failed");
  }
} finally {
  await runner.close();
}
