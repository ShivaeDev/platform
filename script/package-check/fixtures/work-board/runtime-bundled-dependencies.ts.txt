import assert from "node:assert/strict"
import { createRequire } from "node:module"
import manifest from "@shivaedev/work-board/package.json" with { type: "json" }

const installedBoard = createRequire(import.meta.resolve("@shivaedev/work-board/package.json"))
for (const name of [
  "@effect/atom-react",
  "@shivaedev/effect-form",
  "@shivaedev/effect-react",
  "@tanstack/react-router",
  "class-variance-authority",
  "clsx",
  "lucide-react",
  "tailwind-merge",
]) {
  assert(!(name in manifest.dependencies), `${name} must not be a runtime dependency`)
  assert(name in manifest.devDependencies, `${name} must be a build dependency`)
  assert.throws(() => installedBoard.resolve(name), { code: "MODULE_NOT_FOUND" }, `${name} must not be installed`)
}

for (const path of [
  "client/client.ts",
  "client/entry.ts",
  "server/board.ts",
  "server/http/assets.ts",
  "browser/views/now.ts",
  "attention/requestInbox.ts",
  "attention/status.ts",
  "rpc/HandlerOptions.ts",
  "rpc/PageUrl.ts",
  "http/boardRoutes.ts",
  "http/rpcPolicy.ts",
  "browser/document/layout.ts",
  "browser/document/Details.ts",
  "browser/document/sourceGroups.ts",
  "document/answerBody.ts",
  "document/requestTitle.ts",
  "browser/responses/reasonMarkdown.ts",
  "browser/responses/cardContext.ts",
]) {
  assert.throws(
    () => import.meta.resolve(`@shivaedev/work-board/${path}`),
    { code: "ERR_PACKAGE_PATH_NOT_EXPORTED" },
    `${path} must stay internal to Work Board`,
  )
}
