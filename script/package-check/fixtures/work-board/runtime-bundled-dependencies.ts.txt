import assert from "node:assert/strict"
import { createRequire } from "node:module"
import manifest from "@shivaedev/work-board/package.json" with { type: "json" }

const installedBoard = createRequire(import.meta.resolve("@shivaedev/work-board/package.json"))
for (const name of [
  "@base-ui/react",
  "@effect/atom-react",
  "@fontsource-variable/geist",
  "@fontsource-variable/geist-mono",
  "@shivaedev/effect-form",
  "@shivaedev/effect-react",
  "@tanstack/react-router",
  "@tanstack/react-table",
  "class-variance-authority",
  "cmdk",
  "cn",
  "lucide-react",
  "recharts",
  "tw-animate-css",
]) {
  assert(!(name in manifest.dependencies), `${name} must not be a runtime dependency`)
  assert(name in manifest.devDependencies, `${name} must be a build dependency`)
  assert.throws(() => installedBoard.resolve(name), { code: "MODULE_NOT_FOUND" }, `${name} must not be installed`)
}

for (const path of [
  "client/formDraft.ts",
  "client/formStatus.ts",
  "client/react/loadBoardPage.ts",
  "client/responses/retainContext.ts",
]) {
  assert.throws(
    () => import.meta.resolve(`@shivaedev/work-board/${path}`),
    { code: "ERR_PACKAGE_PATH_NOT_EXPORTED" },
    `${path} imports build-only dependencies, so it stays a bundle input`,
  )
}
