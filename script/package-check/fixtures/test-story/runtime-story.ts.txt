import { PassThrough } from "node:stream"
import { genreTag } from "@shivaedev/test-story/genreTag.ts"
import { createVitest } from "vitest/node"

const vitest = await createVitest(
  "test",
  { config: false, include: ["garden.spec.ts"], root: process.cwd(), tags: [genreTag("garden"), genreTag("pond")], tagsFilter: ["garden-story"], watch: false },
  {},
  { stderr: new PassThrough(), stdout: new PassThrough() },
)
const { testModules } = await vitest.start()
await vitest.close()

const outcomes = testModules.flatMap((module) => [
  ...module.errors().map((error) => error.message),
  ...[...module.children.allTests()].map((test) => {
    const result = test.result()
    const headline = result.state === "failed" ? `: ${result.errors[0]?.message.split("\n")[0] ?? ""}` : ""
    return `${test.name} ${result.state}${headline}`
  }),
])
const refusal = [...(testModules[0]?.children.allTests() ?? [])].find((test) => test.name === "grows nothing on stone")?.result()

if (
  outcomes.join("|") !==
    'sprouts every seed passed|grows nothing on stone failed: the trait "the garden is paved" refused to set up the garden: nothing grows on stone|has no fish yet skipped' ||
  refusal?.state !== "failed" ||
  !refusal.errors[0]?.message.includes("✗ given  the garden is paved")
) {
  throw new Error(`The packed story kit misbehaved: ${outcomes.join(" | ")}`)
}

// Vitest marks the process failed for the story that fails on purpose.
process.exitCode = 0
