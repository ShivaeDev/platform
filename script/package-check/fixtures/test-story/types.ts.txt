import { Data, Effect } from "effect"
import { effectStoryKit } from "@shivaedev/test-story/effect/storyKit.ts"
import { type Story, storyKit, type Trait } from "@shivaedev/test-story/storyKit.ts"

interface Garden {
  beds: number
}

class Dry extends Data.TaggedError("Dry") {}

const garden = storyKit({
  after: { dug: () => undefined },
  create: (): Garden => ({ beds: 0 }),
  name: "garden",
  stages: ["dug", "planted"],
  verbs: (state) => ({ gardener: { counts: (): number => state.beds } }),
})
const dug: Trait<Garden, "dug" | "planted"> = garden.trait("dug", "the garden has 2 beds", (state) => {
  state.beds = 2
})
const started: { readonly gardener: { readonly counts: () => number }; readonly story: Story<Garden> } = garden.start(garden.traits(dug))
void started

// @ts-expect-error A trait names one of the kit's stages.
garden.trait("harvested", "the garden is harvested", () => undefined)

const plot = effectStoryKit({
  create: (): Garden => ({ beds: 0 }),
  name: "plot",
  run: {
    maxSteps: 1,
    *step() {
      yield* new Dry()
    },
  },
  stages: ["dug"],
  verbs: (_state, story) => ({ gardener: { waits: () => story.runUntil(() => false) } }),
})
const waited: Effect.Effect<void, Dry> = Effect.flatMap(plot.start(), ({ gardener }) => gardener.waits())
void waited
