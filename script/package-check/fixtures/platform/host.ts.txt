import { Effect, Layer, Option, Schema } from "effect"
import { Rpc, RpcGroup } from "effect/rpc"
import { Migrator } from "effect/sql"
import { appHost, appDatabase, AppChanges } from "@shivaedev/platform/appHost.ts"

const Api = RpcGroup.make(Rpc.make("ping", { success: Schema.String }))
const migrations = { loader: Migrator.fromRecord({}) }
const config = { port: 0 }
const host = appHost({
  group: Api,
  handlers: Api.toLayer({ ping: () => Effect.succeed("pong") }),
  services: Layer.empty,
  migrations,
  sessions: { origin: () => true, provider: { get: () => Effect.succeed(Option.none()) } },
  config,
})
const database = appDatabase({ migrations })
if (!Layer.isLayer(host) || !Layer.isLayer(database) || typeof AppChanges.key !== "string") {
  throw new Error("The packed host entry did not expose its native Layers and live service")
}
