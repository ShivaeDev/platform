import { InternalServerError, ServiceUnavailable } from "@shivaedev/platform/errors/taxonomy.ts"
import { rpcBoundary } from "@shivaedev/platform/server/rpc-server/boundary.ts"
import { nodeSubscriptionSignal } from "@shivaedev/platform/server/node-http/subscriptionSignal.ts"
import { Identity } from "@shivaedev/platform/rpc/identity.ts"
import { makePlatformRuntime } from "@shivaedev/platform/server/runtime/makePlatform.ts"
import { localRpcGuard, loopbackOnly, sameOrigin } from "@shivaedev/platform/server/rpc-server/local.ts"
import { Cause, Context, Effect, Layer, Option } from "effect"
import { HttpServerRequest, HttpServerResponse } from "effect/http"
import { RpcSerialization } from "effect/rpc"

class Greeting extends Context.Service<Greeting, string>()("@consumer/Greeting") {}

const runtime = makePlatformRuntime(Layer.merge(
	Layer.succeed(Greeting, "application"),
	Layer.succeed(Identity, { id: "application" }),
))
const application: string = await runtime.runPromise(Greeting)
const request: string = await runtime.runWithServices(
	Context.make(Greeting, "request"),
	() => runtime.runPromise(Greeting),
)
const identity = await runtime.runWithServices(
	Context.make(Identity, { id: "consumer" }),
	() => runtime.runPromise(Identity),
)
await runtime.dispose()

const source = new AbortController()
const subscription = nodeSubscriptionSignal({ signals: [source.signal] })
source.abort()
subscription.dispose()

const incoming = HttpServerRequest.fromWeb(new Request("http://localhost/rpc", {
	headers: { host: "localhost", origin: "http://localhost", "content-type": "application/ndjson" },
})).modify({ remoteAddress: Option.some("127.0.0.1") })
const guarded = await Effect.runPromise(localRpcGuard(Effect.succeed(HttpServerResponse.empty())).pipe(
	Effect.provideService(HttpServerRequest.HttpServerRequest, incoming),
	Effect.provide(RpcSerialization.layerNdjson),
))
const page = await Effect.runPromise(loopbackOnly(() => Effect.succeed(HttpServerResponse.empty()))(incoming))

if (application !== "application" || request !== "request" || identity.id !== "consumer" || !subscription.signal.aborted || guarded.status !== 204 || page.status !== 204 || !sameOrigin(incoming)) {
	throw new Error("The packed effect-only entries misbehaved")
}

const original = new Error("private storage failure")
const expose = rpcBoundary(InternalServerError, { unexpected: (message) => new InternalServerError({ message }) })
const failure: InternalServerError = await Effect.runPromise(Effect.flip(expose(Effect.fail(original))))
if (failure.message !== "Internal server error" || !Cause.isCause(failure.cause) || Cause.squash(failure.cause) !== original) {
  throw new Error("The packed RPC boundary lost its public rejection or server cause")
}
const unavailable = new ServiceUnavailable({ message: "Storage is unavailable" })
const declared = rpcBoundary(ServiceUnavailable, { unexpected: (message) => new ServiceUnavailable({ message }) })
if (await Effect.runPromise(Effect.flip(declared(Effect.fail(unavailable)))) !== unavailable) {
  throw new Error("The packed RPC boundary replaced a declared rejection")
}
