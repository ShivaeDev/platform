import { RejectionError } from "@shivaedev/effect-trpc/server/rejection.ts"
import { rejectionFormatter } from "@shivaedev/effect-trpc/server/rejectionFormatter.ts"
import { rejectionOf } from "@shivaedev/effect-trpc/rejectionDecoding.ts"
import { rejectionOf as sharedRejectionOf } from "@shivaedev/effect-trpc/rejectionDecoding.ts"
if (rejectionOf !== sharedRejectionOf) throw new Error("The client and shared entries must expose the same decoder")
const error = new RejectionError({ _tag: "NotFound", message: "Gone" })
if (error.code !== "NOT_FOUND") throw new Error("RejectionError lost its taxonomy code: " + error.code)
const shape = rejectionFormatter({ shape: { message: "Gone", code: -32004, data: { code: "NOT_FOUND", httpStatus: 404 } }, error })
const read = rejectionOf(shape)
if (read._tag !== "Some" || read.value._tag !== "NotFound") throw new Error("The client entry did not read the formatted rejection")

for (const entry of ["server/internal/runtime.ts"]) {
  const outcome = await import("@shivaedev/effect-trpc/" + entry).then(
    () => "the import succeeded",
    (error: { readonly code?: string }) => error.code,
  );
  if (outcome !== "ERR_PACKAGE_PATH_NOT_EXPORTED") throw new Error("The packed package exposed " + entry + ": " + outcome);
}
