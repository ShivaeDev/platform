import assert from "node:assert/strict";
import { createServer } from "node:http";
import { Effect, Layer, Option, Schema, Stream } from "effect";
import { FetchHttpClient } from "effect/http";
import { GitHub } from "@shivaedev/github/github/service.ts";
import { list } from "@shivaedev/github/list.ts";
import { read } from "@shivaedev/github/read.ts";
import { write } from "@shivaedev/github/write.ts";
import { GitHubConfig } from "@shivaedev/github/GitHubConfig.ts";
import { GitHubToken } from "@shivaedev/github/GitHubToken.ts";
import { GitHubRateLimited } from "@shivaedev/github/errors.ts";

const server = createServer((request, response) => {
  if (request.url === "/repos/acme/widgets/actions/artifacts/7/zip") {
    response.writeHead(302, { location: `http://${request.headers.host}/storage/archive.zip` });
    response.end();
    return;
  }
  if (request.url === "/storage/archive.zip") {
    response.writeHead(200, { "content-type": "application/zip" });
    response.end(Buffer.from("UEsDBBQAAAAIAAAAIVyk1aTFEgAAABAAAAAKAAAAcmVwb3J0LnR4dHMsKslMS0wuUShKLcgvKuECAFBLAQIUAxQAAAAIAAAAIVyk1aTFEgAAABAAAAAKAAAAAAAAAAAAAACAAQAAAAByZXBvcnQudHh0UEsFBgAAAAABAAEAOAAAADoAAAAAAA==", "base64"));
    return;
  }
  response.writeHead(200, { "content-type": "application/json" });
  response.end(JSON.stringify(request.url === "/items" ? [{ name: "packed" }] : { name: "packed" }));
});
await new Promise<void>((resolve) => server.listen(0, "127.0.0.1", resolve));
try {
  const address = server.address();
  assert(address !== null && typeof address !== "string");
  const layer = GitHub.layer.pipe(Layer.provide(Layer.mergeAll(
    FetchHttpClient.layer,
    Layer.succeed(GitHubToken)(Option.none()),
    Layer.succeed(GitHubConfig)({ apiUrl: `http://127.0.0.1:${address.port}` }),
  )));
  const Item = Schema.Struct({ name: Schema.String });
  const program = Effect.gen(function* () {
    const item: { readonly name: string } = yield* read("/item", Item);
    assert.equal(item.name, "packed");
    assert.deepEqual(yield* list("/items", Item).pipe(Stream.runCollect), [item]);
    assert.deepEqual(yield* write("PATCH", "/item", { name: "packed" }, Item), item);
    assert.deepEqual(yield* (yield* GitHub).observations(), []);
    const files = yield* (yield* GitHub).downloadArtifactArchive({ artifactId: 7, byteLimit: 1024, owner: "acme", repo: "widgets" });
    assert.deepEqual(files, [{ path: "report.txt", bytes: new TextEncoder().encode("Artifact report\n") }]);
  });
  await Effect.runPromise(program.pipe(Effect.provide(layer)));
  assert.equal(typeof GitHubRateLimited, "function");
} finally {
  server.closeAllConnections();
  await new Promise<void>((resolve) => server.close(() => resolve()));
}
