import assert from "node:assert/strict";
import { readdir } from "node:fs/promises";
import { createRequire } from "node:module";

const require = createRequire(import.meta.url);
const packageRequire = createRequire(require.resolve("@shivaedev/agent-pi/pi.ts"));
const installed = await readdir(new URL("./node_modules/.pnpm/", import.meta.url));
for (const sdk of ["@anthropic-ai/claude-agent-sdk", "@openai/codex-sdk", "@opencode-ai/sdk"]) {
	assert.throws(() => packageRequire.resolve(sdk), { code: "MODULE_NOT_FOUND" });
	assert.equal(installed.some((entry) => entry.startsWith(`${sdk.replaceAll("/", "+")}@`)), false, `${sdk} is installed`);
}
