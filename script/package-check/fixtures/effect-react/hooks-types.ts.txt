import { type Cause, Effect, type Option } from "effect";
import * as Atom from "effect/unstable/reactivity/Atom";
import { useAction, useQuery } from "@shivaedev/effect-react";

const query = Atom.make(Effect.succeed({ id: 1, title: "Quarterly report" }));
const action = Atom.fn<{ readonly title: string }>()((input) =>
	input.title.length > 0 ? Effect.succeed({ id: 1, title: input.title }) : Effect.fail("InvalidTitle" as const),
);

export const useExample = () => {
	const result = useQuery(query);
	const data: Option.Option<{ id: number; title: string }> = result.data;
	const mutation = useAction(action);
	const dispatch: (input: { readonly title: string }) => void = mutation.dispatch;
	const failure: Option.Option<Cause.Cause<"InvalidTitle">> = mutation.cause;
	const output: Option.Option<{ id: number; title: string }> = mutation.data;
	// @ts-expect-error The action input remains the declared object shape.
	mutation.dispatch({ title: 1 });
	// @ts-expect-error Query data does not invent fields outside the success type.
	const unrelated: Option.Option<{ missing: number }> = result.data;
	void unrelated;
	return { data, dispatch, failure, output };
};
