import { Effect, Redacted, Scope } from "effect"
import { Migrator, SqlClient } from "effect/unstable/sql"
import { makeEffectIt } from "@shivaedev/effect-test/vitest.ts"
import { PostgresTestDatabase } from "@shivaedev/effect-test-sql/database.ts"
import { makePostgresTestLayer } from "@shivaedev/effect-test-sql/postgres.ts"
import { withTestTransaction } from "@shivaedev/effect-test-sql/withTestTransaction.ts"
import { withIsolatedDatabase } from "@shivaedev/effect-test-sql/withIsolatedDatabase.ts"

const layer = makePostgresTestLayer({
  url: Redacted.make("postgresql://packed-consumer"),
  migrations: { loader: Migrator.fromRecord({}), lockTimeout: "5 seconds" },
  fixtures: Effect.gen(function* () { yield* SqlClient.SqlClient }),
})
const { effectApp } = makeEffectIt({
  layer,
  around: withTestTransaction,
  makeHarness: () => Effect.map(SqlClient.SqlClient, sql => ({ sql })),
})
effectApp("retains SQL and harness types", function* ({ sql }) {
  const client: SqlClient.SqlClient = sql
  void client
  yield* Scope.Scope
  yield* withIsolatedDatabase(Effect.gen(function* () {
    const database = yield* PostgresTestDatabase
    const url: Redacted.Redacted<string> = database.url
    void url
    yield* SqlClient.SqlClient
  }))
})
