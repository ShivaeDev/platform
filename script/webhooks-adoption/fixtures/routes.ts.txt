import { Effect, Layer } from "effect";
import { HttpRouter, HttpServerResponse } from "effect/http";
import { SqlClient } from "effect/sql";
import { webhookRoute } from "@shivaedev/webhooks/server/route.ts";
import { sources, WebhookJobs } from "#server/webhooks.ts";

export const webhookRoutes = Layer.unwrap(
	Effect.gen(function* () {
		const sql = yield* SqlClient.SqlClient;
		const jobs = yield* WebhookJobs;
		return Layer.merge(
			webhookRoute({ basePath: "/webhooks", jobs: WebhookJobs, sources }).pipe(
				Layer.provide(Layer.succeed(WebhookJobs, jobs)),
				Layer.provide(Layer.succeed(SqlClient.SqlClient, sql)),
			),
			HttpRouter.add("GET", "/indexed-issues", () =>
				Effect.map(sql<{ number: number }>`select number from indexed_issues order by number`, (rows) =>
					HttpServerResponse.jsonUnsafe(rows.map(({ number }) => number)),
				),
			),
		);
	}),
);
